```cpp

// 记录下所有禁用的功能的信息
void ABasePlayerState::CollectPgosFunctionBan(FPgosPlayerKVDataGroup& PgosKvDatas) const  
{  
    const auto& PlayerID = PgosKvDatas.player_id;  
    auto& kvdata = PgosKvDatas.kvdata;  
  
    TSharedPtr<FJsonObject> BanJson = MakeShareable(new FJsonObject);  
    for (const auto& BanStatePair : PlayerProfileData->m_banStates)  
    {  
       TSharedPtr<FJsonObject> BanItem = MakeShareable(new FJsonObject);  
       BanItem->SetNumberField(TEXT("end_time"), BanStatePair.Value.EndTimeStamp);  
       BanItem->SetStringField(TEXT("hint"), BanStatePair.Value.Hint);  
       BanItem->SetStringField(TEXT("reason"), BanStatePair.Value.BanReason);  
       BanJson->SetObjectField(FString::FromInt(BanStatePair.Key), BanItem);  
    }  
  
    FString BanJsonStr;  
    TSharedRef<TJsonWriter<>> Writer = TJsonWriterFactory<>::Create(&BanJsonStr);  
    FJsonSerializer::Serialize(BanJson.ToSharedRef(), Writer);  
  
    kvdata.Emplace(TEXT("FunctionBan"), BanJsonStr);  
  
    UE_LOG(ServerPgosApi, Log, TEXT("CollectPgosFunctionBan, BanCount=%d, PlayerID=%s"),  
       PlayerProfileData->m_banStates.Num(), *PlayerID);  
}
```

	       OutUserInfo.BanEndTimes.Empty();

	          {  
	             OutUserInfo.BanEndTimes.Add(BanType, static_cast<int64>(BanItem->GetNumberField(TEXT("end_time"))));  
	          }  
	       }  
	    }  
	}
